# MobilityAI — Python CV processor dependencies.
#
# Install:
#   python -m venv venv && source venv/bin/activate
#   pip install -r requirements.txt
#
# Python 3.11 is the supported target — MediaPipe publishes wheels for it and
# it is what the Docker image pins. 3.12 generally works; 3.13+ has no
# MediaPipe wheel at time of writing.

# ── Web layer ────────────────────────────────────────────────────
flask>=3.0,<4
flask-cors>=4.0            # app.py imports this — required, was missing before
gunicorn>=21.2             # production WSGI server (Flask's dev server is not)

# ── CV / ML core ─────────────────────────────────────────────────
mediapipe>=0.10.14,<0.11
# Headless build: no GUI/X11 system libraries needed (nothing in this codebase
# calls imshow/waitKey), and `contrib` supplies cv2.TrackerKCF_create used by
# utils/bar_tracker.py. Do NOT also install opencv-python or
# opencv-contrib-python — the three variants conflict with each other.
opencv-contrib-python-headless>=4.8,<5
# MediaPipe 0.10.x is only reliably compatible with the NumPy 1.x ABI.
numpy>=1.26,<2.0

# ── Signal processing / analysis ─────────────────────────────────
scipy>=1.11           # Savitzky-Golay smoothing (utils/signal_filters.py)
filterpy>=1.4.5       # Kalman 1-D for bar position / velocity
scikit-learn>=1.3     # Regression / future fault classifiers
dtaidistance>=2.3     # Fast DTW for rep-template similarity
PyYAML>=6.0           # Eval harness annotation files

# ── Optional (lazy-loaded) ───────────────────────────────────────
# Uncomment to enable the YOLOv8 plate-detection fallback.
# ultralytics>=8.0
